% SUBSECTIONS WITH COUNTERS

% The aim here is to have counter images at the left margin and a subsection
% describing the counter indented appropriately.

\usepackage{needspace}

\makeatletter

\newcommand{\vspacelines}[1]{\vspace{#1\baselineskip}}

\newcommand{\@subsectionwithcounters}[3]{%
    \Needspace{#1\baselineskip}%
    \@unindentedsubsection[#2]{\makebox[0em][r]{#3\hspace{0.5em}}#2}
}

\newcommand{\subsectionwithcounter}[2]{%
    \@subsectionwithcounters{4}{#2}{%
        \makebox[0em][r]{\raisebox{-3.8em}[0em][0em]{\includegraphics[width=4.5em]{counters/#1.pdf}}}%
    }
}

\newcommand{\subsectionwithtwocounters}[3]{%
    \@subsectionwithcounters{8}{#3}{%
        \makebox[0em][r]{\raisebox{-3.8em}[0em][0em]{\includegraphics[width=4.5em]{counters/#1.pdf}}}%
        \makebox[0em][r]{\raisebox{-8.8em}[0em][0em]{\includegraphics[width=4.5em]{counters/#2.pdf}}}%
    }
}

\newcommand{\subsectionwiththreecounters}[4]{%
    \@subsectionwithcounters{12}{#4}{%
        \makebox[0em][r]{\raisebox{-3.8em}[0em][0em]{\includegraphics[width=4.5em]{counters/#1.pdf}}}%
        \makebox[0em][r]{\raisebox{-8.8em}[0em][0em]{\includegraphics[width=4.5em]{counters/#2.pdf}}}%
        \makebox[0em][r]{\raisebox{-13.8em}[0em][0em]{\includegraphics[width=4.5em]{counters/#3.pdf}}}%
    }
}

\newcommand{\subsectionwithfourcounters}[5]{%
    \@subsectionwithcounters{16}{#5}{%
        \makebox[0em][r]{\raisebox{-3.8em}[0em][0em]{\includegraphics[width=4.5em]{counters/#1.pdf}}}%
        \makebox[0em][r]{\raisebox{-8.8em}[0em][0em]{\includegraphics[width=4.5em]{counters/#2.pdf}}}%
        \makebox[0em][r]{\raisebox{-13.8em}[0em][0em]{\includegraphics[width=4.5em]{counters/#3.pdf}}}%
        \makebox[0em][r]{\raisebox{-18.8em}[0em][0em]{\includegraphics[width=4.5em]{counters/#4.pdf}}}%
    }
}

\newenvironment{withcounters}{%
    \setlength{\leftskip}{5em}%
    \ignorespaces\@ifnextchar\par{\@gobble}{}
}{%
    \setlength{\leftskip}{0em}
}

\makeatother

% RULES SECTIONS

% Rule sections are built upon normal section.

\makeatletter

\newcounter{rulesection}
\newcommand{\rulesection}[1]{%
    \stepcounter{rulesection}
    \section{Rule \arabic{rulesection}: #1}
}

\newcounter{rulesubsection}[rulesection]
\newcommand{\rulesubsection}[1]{%
    \stepcounter{rulesubsection}
    \subsection{\Alph{rulesubsection}. #1}
}

% Examples: in 40.B and 40.C.
\newcounter{rulesubsubsection}[rulesubsection]
\newcommand{\rulesubsubsection}[1]{%
    \stepcounter{rulesubsubsection}
    \subsubsection{\arabic{rulesubsubsection}. #1}
}

\newcommand{\ruleparagraph}[1]{
    \paragraph{#1}
}

\newcommand{\advancedrule}{
    \@startsection{section}{1}%
    {0pt}%
    {1.0ex plus 0.3ex minus 0.1ex}%
    {0.1ex plus 0.1ex minus 0.1ex}%
    {\normalfont\normalsize\bfseries\raggedright}%
    *{Advanced Rule}
}

\makeatother
